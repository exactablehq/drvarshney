"use client";

import { useState } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { Camera, Sparkles, X, Maximize2, ChevronLeft, ChevronRight } from "lucide-react";

export default function Gallery() {
  const [selectedItemIndex, setSelectedItemIndex] = useState<number | null>(null);

  const selectedItem = selectedItemIndex !== null ? galleryItems[selectedItemIndex] : null;

  const handleNext = () => {
    if (selectedItemIndex !== null) {
      setSelectedItemIndex((selectedItemIndex + 1) % galleryItems.length);
    }
  };

  const handlePrev = () => {
    if (selectedItemIndex !== null) {
      setSelectedItemIndex((selectedItemIndex - 1 + galleryItems.length) % galleryItems.length);
    }
  };

  return (
    <section id="gallery" className="relative pt-12 pb-20 lg:pt-16 lg:pb-28 overflow-hidden bg-[#090611] border-t border-[#35063e]/20 flex flex-col justify-center">
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 relative z-10 w-full flex flex-col items-center">
        
        {/* Header */}
        <div className="text-center max-w-3xl mx-auto flex flex-col items-center mb-10 lg:mb-14">
          <div className="inline-flex items-center gap-1.5 px-3.5 py-1.5 rounded-full bg-[#0a0516]/65 border border-purple-500/30 text-xs font-semibold text-purple-300 tracking-wider uppercase backdrop-blur-[10px] mb-6">
            <Camera className="w-3.5 h-3.5 text-purple-400" />
            <span>Clinic Showcase</span>
          </div>

          <h2 className="text-3xl sm:text-4xl lg:text-5xl font-sans font-extrabold text-white tracking-tight leading-[1.15] mb-6">
            Visual Excellence. <span className="beautiful-smiles-glow">Beautiful Smiles.</span>
          </h2>
        </div>

        {/* Static Photo Grid */}
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6 w-full">
          {galleryItems.map((item, idx) => (
            <div
              key={item.id}
              onClick={() => setSelectedItemIndex(idx)}
              className="group relative rounded-2xl sm:rounded-3xl overflow-hidden aspect-[4/3] bg-[#0e071b] border border-purple-500/20 cursor-pointer shadow-[0_8px_25px_rgba(0,0,0,0.5)] transition-all duration-500 hover:border-purple-500/50 hover:shadow-[0_12px_35px_rgba(168,85,247,0.25)]"
            >
              <img src={item.image} alt={item.title} className="w-full h-full object-cover transition-transform duration-700 ease-out group-hover:scale-110" />
              <div className="absolute inset-0 bg-gradient-to-t from-[#090611]/90 via-[#090611]/25 to-transparent opacity-0 group-hover:opacity-100 transition-opacity duration-300 flex flex-col justify-between p-5">
                <span className="px-3 py-1 rounded-full text-[10px] font-semibold uppercase tracking-wider bg-[#0a0516]/80 border border-purple-500/40 text-purple-300 backdrop-blur-md self-start">
                  {item.tag}
                </span>
                <div className="flex items-end justify-between gap-3">
                  <div>
                    <h3 className="text-base sm:text-lg font-bold text-white leading-snug">{item.title}</h3>
                  </div>
                  <div className="w-10 h-10 rounded-full bg-purple-600/80 text-white flex items-center justify-center shrink-0 border border-purple-400/50">
                    <Maximize2 className="w-4 h-4" />
                  </div>
                </div>
              </div>
            </div>
          ))}
        </div>
      </div>

      {/* Enlarged Full-Size Photo Modal */}
      <AnimatePresence>
        {selectedItem && (
          <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-6 bg-black/95 backdrop-blur-xl">
            <button onClick={() => setSelectedItemIndex(null)} className="absolute top-5 right-5 z-50 p-3 rounded-full bg-black/70 border border-white/20 text-white hover:bg-purple-600/60">
              <X className="w-6 h-6" />
            </button>
            <button onClick={handlePrev} className="absolute left-3 sm:left-6 top-1/2 -translate-y-1/2 z-50 p-3.5 rounded-full bg-black/70 border border-white/20 text-white hover:bg-purple-600/60">
              <ChevronLeft className="w-6 h-6" />
            </button>
            <button onClick={handleNext} className="absolute right-3 sm:right-6 top-1/2 -translate-y-1/2 z-50 p-3.5 rounded-full bg-black/70 border border-white/20 text-white hover:bg-purple-600/60">
              <ChevronRight className="w-6 h-6" />
            </button>
            <div className="relative max-w-6xl max-h-[85vh] z-40 flex flex-col items-center justify-center overflow-hidden rounded-2xl border border-purple-500/30 bg-black">
              <img src={selectedItem.image} alt={selectedItem.title} className="max-w-full max-h-[75vh] object-contain" />
              <div className="w-full bg-[#0a0516]/90 px-6 py-4 flex flex-col sm:flex-row items-center justify-between gap-2">
                <h3 className="text-lg font-bold text-white">{selectedItem.title}</h3>
                <span className="px-3 py-1 rounded-full text-xs font-semibold uppercase bg-purple-500/20 text-purple-300">{selectedItem.tag}</span>
              </div>
            </div>
          </div>
        )}
      </AnimatePresence>
    </section>
  );
}
